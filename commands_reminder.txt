Docker Commands
---

# to build the image (shared ctx must be injected)
# run inside src/animacharacter_server/
docker build --build-context shared=../shared_libs -t ac-engine:latest .

# run the image in deploy mode
docker run -it --rm --network host ac-engine:latest

# run the image and start the engine immediately
docker run -it --rm --network host ac-engine:latest start

# run image and mount a custom plugins folder
docker run -it --rm --network host -v ./custom_plugins:/app/plugins ac-engine:latest
