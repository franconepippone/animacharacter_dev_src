# to build the image
docker build -t ros2-jazzy-dev .

# run the image in dev mode (mount local workspace)
docker run -it --rm   --net=host   -v $(pwd)/ros2_ws:/ros2_ws   ros2-jazzy-dev

# run the image in deploy mode (all the files are in the image)
docker run -it --rm   --net=host     ros2-jazzy-dev
